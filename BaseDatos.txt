CREATE DATABASE InventarioFerreteriaDB
GO

USE InventarioFerreteriaDB;
GO

CREATE TABLE Categorias
(
    IdCategoria INT IDENTITY(1,1) PRIMARY KEY,
    Nombre NVARCHAR(100) NOT NULL,
    Descripcion NVARCHAR(250) NULL,
    CONSTRAINT UQ_Categorias_Nombre UNIQUE (Nombre)
);
GO

CREATE TABLE Marcas
(
    IdMarca INT IDENTITY(1,1) PRIMARY KEY,
    Nombre NVARCHAR(100) NOT NULL,
    CONSTRAINT UQ_Marcas_Nombre UNIQUE (Nombre)
);
GO

CREATE TABLE Productos
(
    IdProducto INT IDENTITY(1,1) PRIMARY KEY,
    Nombre NVARCHAR(150) NOT NULL,
    Descripcion NVARCHAR(300) NULL,
    IdCategoria INT NOT NULL,
    Activo BIT NOT NULL DEFAULT 1,
    CONSTRAINT FK_Productos_Categorias
        FOREIGN KEY (IdCategoria)
        REFERENCES Categorias(IdCategoria)
);
GO

CREATE TABLE VariantesProducto
(
    IdVariante INT IDENTITY(1,1) PRIMARY KEY,
    IdProducto INT NOT NULL,
    IdMarca INT NULL,
    SKU NVARCHAR(50) NOT NULL,
    Color NVARCHAR(50) NULL,
    Tamano NVARCHAR(50) NULL,
    Material NVARCHAR(80) NULL,
    Medida NVARCHAR(80) NULL,
    Presentacion NVARCHAR(100) NULL,
    PrecioCompra DECIMAL(10,2) NOT NULL,
    PrecioVenta DECIMAL(10,2) NOT NULL,
    StockMinimo INT NOT NULL DEFAULT 5,
    Activo BIT NOT NULL DEFAULT 1,
    CONSTRAINT UQ_VariantesProducto_SKU UNIQUE (SKU),
    CONSTRAINT FK_VariantesProducto_Productos
        FOREIGN KEY (IdProducto)
        REFERENCES Productos(IdProducto),
    CONSTRAINT FK_VariantesProducto_Marcas
        FOREIGN KEY (IdMarca)
        REFERENCES Marcas(IdMarca)
);
GO

CREATE TABLE Sucursales
(
    IdSucursal INT IDENTITY(1,1) PRIMARY KEY,
    Nombre NVARCHAR(120) NOT NULL,
    Direccion NVARCHAR(250) NULL,
    Telefono NVARCHAR(20) NULL,
    Activo BIT NOT NULL DEFAULT 1
);
GO

CREATE TABLE Proveedores
(
    IdProveedor INT IDENTITY(1,1) PRIMARY KEY,
    Nombre NVARCHAR(150) NOT NULL,
    Telefono NVARCHAR(20) NULL,
    Correo NVARCHAR(120) NULL,
    Direccion NVARCHAR(250) NULL,
    TiempoEntregaDias INT NOT NULL DEFAULT 3,
    Activo BIT NOT NULL DEFAULT 1
);
GO

CREATE TABLE Inventarios
(
    IdInventario INT IDENTITY(1,1) PRIMARY KEY,
    IdVariante INT NOT NULL,
    IdSucursal INT NOT NULL,
    Cantidad INT NOT NULL DEFAULT 0,
    CONSTRAINT FK_Inventarios_VariantesProducto
        FOREIGN KEY (IdVariante)
        REFERENCES VariantesProducto(IdVariante),
    CONSTRAINT FK_Inventarios_Sucursales
        FOREIGN KEY (IdSucursal)
        REFERENCES Sucursales(IdSucursal),
    CONSTRAINT UQ_Inventarios_VarianteSucursal
        UNIQUE (IdVariante, IdSucursal)
);
GO

CREATE TABLE EntradasInventario
(
    IdEntrada INT IDENTITY(1,1) PRIMARY KEY,
    IdVariante INT NOT NULL,
    IdSucursal INT NOT NULL,
    IdProveedor INT NULL,
    Cantidad INT NOT NULL,
    CostoUnitario DECIMAL(10,2) NOT NULL,
    Fecha DATETIME NOT NULL DEFAULT GETDATE(),
    CONSTRAINT FK_Entradas_VariantesProducto
        FOREIGN KEY (IdVariante)
        REFERENCES VariantesProducto(IdVariante),
    CONSTRAINT FK_Entradas_Sucursales
        FOREIGN KEY (IdSucursal)
        REFERENCES Sucursales(IdSucursal),
    CONSTRAINT FK_Entradas_Proveedores
        FOREIGN KEY (IdProveedor)
        REFERENCES Proveedores(IdProveedor)
);
GO

CREATE TABLE Ventas
(
    IdVenta INT IDENTITY(1,1) PRIMARY KEY,
    IdSucursal INT NOT NULL,
    Fecha DATETIME NOT NULL DEFAULT GETDATE(),
    Total DECIMAL(10,2) NOT NULL,
    CONSTRAINT FK_Ventas_Sucursales
        FOREIGN KEY (IdSucursal)
        REFERENCES Sucursales(IdSucursal)
);
GO

CREATE TABLE DetalleVentas
(
    IdDetalleVenta INT IDENTITY(1,1) PRIMARY KEY,
    IdVenta INT NOT NULL,
    IdVariante INT NOT NULL,
    Cantidad INT NOT NULL,
    PrecioUnitario DECIMAL(10,2) NOT NULL,
    Subtotal DECIMAL(10,2) NOT NULL,
    CONSTRAINT FK_DetalleVentas_Ventas
        FOREIGN KEY (IdVenta)
        REFERENCES Ventas(IdVenta),
    CONSTRAINT FK_DetalleVentas_VariantesProducto
        FOREIGN KEY (IdVariante)
        REFERENCES VariantesProducto(IdVariante)
);
GO

CREATE TABLE MovimientosInventario
(
    IdMovimiento INT IDENTITY(1,1) PRIMARY KEY,
    IdVariante INT NOT NULL,
    IdSucursal INT NOT NULL,
    TipoMovimiento NVARCHAR(20) NOT NULL,
    Cantidad INT NOT NULL,
    Fecha DATETIME NOT NULL DEFAULT GETDATE(),
    Descripcion NVARCHAR(250) NULL,
    CONSTRAINT FK_Movimientos_VariantesProducto
        FOREIGN KEY (IdVariante)
        REFERENCES VariantesProducto(IdVariante),
    CONSTRAINT FK_Movimientos_Sucursales
        FOREIGN KEY (IdSucursal)
        REFERENCES Sucursales(IdSucursal)
);
GO

CREATE TABLE Alertas
(
    IdAlerta INT IDENTITY(1,1) PRIMARY KEY,
    IdVariante INT NOT NULL,
    IdSucursal INT NOT NULL,
    TipoAlerta NVARCHAR(50) NOT NULL,
    Mensaje NVARCHAR(250) NOT NULL,
    Fecha DATETIME NOT NULL DEFAULT GETDATE(),
    Estado NVARCHAR(20) NOT NULL DEFAULT 'Pendiente',
    CONSTRAINT FK_Alertas_VariantesProducto
        FOREIGN KEY (IdVariante)
        REFERENCES VariantesProducto(IdVariante),
    CONSTRAINT FK_Alertas_Sucursales
        FOREIGN KEY (IdSucursal)
        REFERENCES Sucursales(IdSucursal)
);
GO

CREATE TABLE Recomendaciones
(
    IdRecomendacion INT IDENTITY(1,1) PRIMARY KEY,
    IdVariante INT NOT NULL,
    IdSucursal INT NOT NULL,
    StockActual INT NOT NULL,
    PromedioVentaDiaria DECIMAL(10,2) NOT NULL,
    DiasRestantes DECIMAL(10,2) NOT NULL,
    CantidadRecomendada INT NOT NULL,
    Fecha DATETIME NOT NULL DEFAULT GETDATE(),
    CONSTRAINT FK_Recomendaciones_VariantesProducto
        FOREIGN KEY (IdVariante)
        REFERENCES VariantesProducto(IdVariante),
    CONSTRAINT FK_Recomendaciones_Sucursales
        FOREIGN KEY (IdSucursal)
        REFERENCES Sucursales(IdSucursal)
);
GO
